/*
    Item scripts. Display names come from Translate/<LANG>/ItemName.json and tooltips from
    Tooltip.json. Icons are EnhancedIrrigation_<Item>, packed as Item_EnhancedIrrigation_<Item>.

    base:ignorezombiedensity keeps the zombie density loot bonus, which is added to every entry
    whatever its weight, from swamping the pamphlet's small loot weights
    (server/EnhancedIrrigation/Items/Distributions.lua).
*/
module EnhancedIrrigation
{
    imports { Base }

    /*
        A University of Kentucky extension leaflet on storing and moving farm water: the Tank
        Stand (a build-menu recipe) and plumbing tanks together (a knowledge-only recipe,
        Recipes.txt). The skills teach both too. Drawn with vanilla's magazine models, as the
        game's own recipe magazines are.
    */
    item CisternsForKentucky
    {
        DisplayCategory = RecipeResource,
        ItemType = base:literature,
        Weight = 0.1,
        Icon = EnhancedIrrigation_CisternsForKentucky,
        StaticModel = Magazine,
        WorldStaticModel = MagazineGround,
        BoredomChange = -10,
        StressChange = -5,
        LearnedRecipes = EnhancedIrrigation_TankStand;EnhancedIrrigation_PlumbToTank,
        Tooltip = Tooltip_EnhancedIrrigation_CisternsForKentucky,
        OnCreate = ItemCodeOnCreate.onCreateRecipeMagazine,
        Tags = base:magazine;base:ignorezombiedensity,
    }
}
