/*

    Tooltip.json. Icons are EnhancedIrrigation_<Item>, packed as Item_EnhancedIrrigation_<Item>.

    whatever its weight, from swamping the pamphlet's small loot weights
    (server/EnhancedIrrigation/Items/Distributions.lua).
*/
module EnhancedIrrigation
{

    /*

        Tank Stand, the Wooden and Iron Hand Pumps and the Barrel and Brick Filters (build-menu
        recipes) and plumbing tanks together (a knowledge-only recipe, Recipes.txt). The skills
        teach them too. Drawn with vanilla's magazine models, as the game's own recipe magazines
        are.
    */
    item CisternsForKentucky
    {
        DisplayCategory = RecipeResource,
        ItemType = base:literature,
        Weight = 0.1,
        Icon = EnhancedIrrigation_CisternsForKentucky,
        StaticModel = Magazine,
        WorldStaticModel = MagazineGround,
        BoredomChange = -10,
        StressChange = -5,
        LearnedRecipes = EnhancedIrrigation_TankStand;EnhancedIrrigation_PlumbToTank;EnhancedIrrigation_WoodenHandPump;EnhancedIrrigation_IronHandPump;EnhancedIrrigation_BarrelFilter;EnhancedIrrigation_BrickFilter,
        Tooltip = Tooltip_EnhancedIrrigation_CisternsForKentucky,
        OnCreate = ItemCodeOnCreate.onCreateRecipeMagazine,
        Tags = base:magazine;base:ignorezombiedensity,
    }

    /*
        The Filter Crock, shaped at a pottery wheel and fired in a kiln (Recipes.txt), set in a
        bucket from the build menu (Entities.txt). Drawn with vanilla's clay jar models, as

        StaticModel = ClayJar,
        WorldStaticModel = ClayJar,
        Tooltip = Tooltip_EnhancedIrrigation_FilterCrockItem,
        ResearchableRecipes = EnhancedIrrigation_ShapeFilterCrock;EnhancedIrrigation_FireFilterCrock;EnhancedIrrigation_FilterCrock,
    }
}
