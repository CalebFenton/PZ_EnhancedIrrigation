/*
    craftRecipes, in module Base with the mod's prefix: magazines and research look recipes up by
    bare name, which resolves only in Base.

    EnhancedIrrigation_PlumbToTank and EnhancedIrrigation_SalvagePump are knowledge, not crafts:
    the tank menu offers Plumb to Tank, and an appliance's menu Salvage Pump, once the character
    knows it (isRecipeActuallyKnown). Reading Cisterns for Kentucky teaches Plumb to Tank, and
    vanilla's AutoLearn check (ScriptManager.checkAutoLearn, at world load and on every level up)
    learns either from the skills; shared/EnhancedIrrigation/Recipes.lua checks them too and sends
    a multiplayer server's lesson to the client. They have nothing to craft and are never offered:
    - their only tag is one no crafting bench or window asks for, so no recipe query finds them and
      they need a workstation that doesn't exist (CraftRecipe.requiresSpecificWorkstation);
    - the crafting window that lists every recipe ("*", the main window and the magazines'
      recipe lists) drops them through OnAddToMenu, a plain global
      (shared/EnhancedIrrigation/Recipes.lua).
*/
module Base
{
    craftRecipe EnhancedIrrigation_PlumbToTank
    {
        icon = PipeWrench,
        Tags = EnhancedIrrigationKnowledge,
        NeedToBeLearn = true,
        AutoLearnAny = Woodwork:4;MetalWelding:3,
        OnAddToMenu = EnhancedIrrigation_NeverInMenu,
    }

    craftRecipe EnhancedIrrigation_SalvagePump
    {
        icon = Screwdriver,
        Tags = EnhancedIrrigationKnowledge,
        NeedToBeLearn = true,
        AutoLearnAny = Electricity:2,
        OnAddToMenu = EnhancedIrrigation_NeverInMenu,
    }

    /*
        A Utility Pump made from a Blower Fan: the fan's motor drives an impeller cut and welded
        from sheet steel in a screwed housing, wired and fitted with a hose. Too much to hold and
        weld at once, so it needs a table or another surface in reach (AnySurfaceCraft), as
        Renewable Energy's welded battery boxes do; its torch and rods are taken by uses. It comes
        out at a lower ceiling than a found one (Pumps.onMadePump). Learned at Electricity 6 or by
        researching a Utility Pump.
    */
    craftRecipe EnhancedIrrigation_MakeUtilityPump
    {
        timedAction = Welding,
        time = 300,
        Tags = AnySurfaceCraft;Welding,
        category = Electrical,
        NeedToBeLearn = true,
        AutoLearnAny = Electricity:6,
        SkillRequired = Electricity:3;MetalWelding:2,
        xpAward = Electricity:30;MetalWelding:15,
        Tooltip = Tooltip_EnhancedIrrigation_MakeUtilityPump,
        OnCreate = EnhancedIrrigation.Pumps.onMadePump,
        inputs
        {
            item 1 tags[base:weldingmask] mode:keep,
            item 1 tags[base:screwdriver] mode:keep flags[MayDegradeLight],
            item 2 [Base.BlowTorch],
            item 1 [Base.WeldingRods],
            item 1 [Base.BlowerFan],
            item 2 [Base.SmallSheetMetal],
            item 4 [Base.Screws],
            item 2 [Base.ElectricWire],
            item 1 [Base.RubberHose],
        }
        outputs
        {
            item 1 EnhancedIrrigation.UtilityPump,
        }
    }

    /*
        A Lawn Sprinkler made from pipe: a length of iron pipe drilled for a riser, an empty
        gardening spray's nozzle screwed on as its head. Made in hand. It comes out at a lower
        ceiling than a found one (Sprinklers.onMade). Learned at Welding 4 or by researching a Lawn
        Sprinkler, from Welding 1, two-thirds of its skill as vanilla's research asks.
    */
    craftRecipe EnhancedIrrigation_MakePipeSprinkler
    {
        timedAction = Making,
        time = 150,
        Tags = InHandCraft,
        category = Farming,
        NeedToBeLearn = true,
        AutoLearnAny = MetalWelding:4,
        SkillRequired = MetalWelding:2,
        xpAward = MetalWelding:15,
        Tooltip = Tooltip_EnhancedIrrigation_MakePipeSprinkler,
        OnCreate = EnhancedIrrigation.Sprinklers.onMade,
        inputs
        {
            item 1 [Base.HandDrill] mode:keep flags[MayDegradeLight],
            item 1 tags[base:pliers] mode:keep flags[MayDegradeLight],
            item 1 [Base.MetalPipe],
            item 1 [Base.GardeningSprayEmpty],
            item 2 [Base.Screws],
        }
        outputs
        {
            item 1 EnhancedIrrigation.LawnSprinkler,
        }
    }

    /*
        The Wind-up Valve: a kitchen timer's spring and dial screwed to a length of rubber hose it
        pinches shut as it runs down. Anyone can make one. At a table or another surface, as
        vanilla's own timer crafts are (MakeTimer).
    */
    craftRecipe EnhancedIrrigation_MakeWindUpValve
    {
        timedAction = MakingElectrical,
        time = 100,
        Tags = AnySurfaceCraft,
        category = Farming,
        Tooltip = Tooltip_EnhancedIrrigation_MakeWindUpValve,
        inputs
        {
            item 1 tags[base:screwdriver] mode:keep flags[MayDegradeLight],
            item 1 [Base.Timer],
            item 2 [Base.Screws],
            item 1 [Base.RubberHose],
        }
        outputs
        {
            item 1 EnhancedIrrigation.WindUpValve,
        }
    }

    /*
        The Timer Valve: an alarm clock wired to switch a salvaged Water Valve, which opens the tap
        or the Pump House's power at its hour each day. At a table or another surface, as
        vanilla's own timer crafts are. Learned at Electricity 4 or by researching a Water Valve or
        a Sprinkler Timer, from Electricity 1.
    */
    craftRecipe EnhancedIrrigation_MakeTimerValve
    {
        timedAction = MakingElectrical,
        time = 150,
        Tags = AnySurfaceCraft,
        category = Electrical,
        NeedToBeLearn = true,
        AutoLearnAny = Electricity:4,
        SkillRequired = Electricity:2,
        xpAward = Electricity:10,
        Tooltip = Tooltip_EnhancedIrrigation_MakeTimerValve,
        inputs
        {
            item 1 tags[base:screwdriver] mode:keep flags[MayDegradeLight],
            item 1 tags[base:pliers] mode:keep flags[MayDegradeLight],
            item 1 tags[base:pipewrench] mode:keep flags[MayDegradeLight],
            item 1 [Base.AlarmClock2],
            item 1 [EnhancedIrrigation.WaterValve],
            item 2 [Base.ElectricWire],
            item 1 [Base.ElectronicsScrap],
        }
        outputs
        {
            item 1 EnhancedIrrigation.TimerValve,
        }
    }

    /*
        The Filter Crock, made as vanilla's clay jar is: shaped at a pottery wheel, with the water
        from any container kept, then fired in a kiln. The charcoal crushed into the clay burns
        out in the firing and leaves the pores the water seeps through. Both are learned together
        with setting the crock up (Entities.txt): by the Pottery level their AutoLearn names, or by
        researching a Clay Jar (shared/EnhancedIrrigation/Recipes.lua adds that at load) or a
        Filter Crock.
    */
    craftRecipe EnhancedIrrigation_ShapeFilterCrock
    {
        timedAction = Craft_PotteryWheel,
        time = 150,
        Tags = PotteryWheel,
        category = Pottery,
        NeedToBeLearn = true,
        AutoLearnAny = Pottery:5,
        SkillRequired = Pottery:4,
        xpAward = Pottery:40,
        Tooltip = Tooltip_EnhancedIrrigation_ShapeFilterCrock,
        inputs
        {
            item 4 [Base.Clay],
            item 2 tags[base:charcoal],
            item 1 [*] mode:keep flags[ItemIsFluid;HandcraftOnly],
            -fluid 0.5 categories[Water] mode:mixture,
        }
        outputs
        {
            item 1 EnhancedIrrigation.FilterCrockUnfired,
        }
    }

    craftRecipe EnhancedIrrigation_FireFilterCrock
    {
        timedAction = Making_With_Kiln,
        time = 20,
        Tags = KilnSmall;KilnLarge,
        category = Pottery,
        NeedToBeLearn = true,
        AutoLearnAny = Pottery:5,
        SkillRequired = Pottery:4,
        inputs
        {
            item 1 [EnhancedIrrigation.FilterCrockUnfired],
            item 1 tags[base:startfire],
            item 1 tags[base:log;base:charcoal],
        }
        outputs
        {
            item 1 EnhancedIrrigation.FilterCrock,
        }
    }
}
