/*
    The mod's sounds, each a vanilla event under a name of its own, so the event can change without
    the code that plays it.
*/
module Base
{
    /* A wooden hand pump's handle squealing on its dry pivot as it is worked: a wooden door's creak,
       the nearest the game has. */
    sound EnhancedIrrigationPumpSqueal
    {
        category = Object,
        clip
        {
            event = Object/WoodDoor/Creak,
        }
    }

    /* A running Pump House's motor: the hum of a filling station's pump, the nearest the game has,
       looped on each client while the house runs (client/EnhancedIrrigation/Pumps/PumpClient.lua). */
    sound EnhancedIrrigationPumpHouseRunning
    {
        category = Object,
        loop = true,
        clip
        {
            event = World/Object/GasPump,
        }
    }
}
